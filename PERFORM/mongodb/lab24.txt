1. Display distinct city.
db.students.aggregate([{$group:{_id:"$CITY"}}])

2. Display city wise count of number of students.
db.students.aggregate([{$group:{_id:"$CITY",Totalstudent:{$sum:1}}}])

3. Display sum of fees in your collection.
db.students.aggregate([{$group:{_id:null,Totalfess:{$sum:"$FEES"}}}])

4. Display average of fees in your document.
db.students.aggregate([{$group:{_id:null,avgfess:{$avg:"$FEES"}}}])

5. Display maximum and minimum salary of your document.
db.students.aggregate([{$group:{_id:null,minfess:{$min:"$FEES"},maxfees:{$max:"$FEES"}}}])

6. Display city wise total salary in your collection
db.students.aggregate([{$group:{_id:"$CITY",TotalFees:{$sum:"$FEES"}}}])

7. Display gender wise maximum salary in your collection.
db.students.aggregate([{$group:{_id:"$GENDER",TotalFees:{$max:"$FEES"}}}])

8. Display city wise maximum and minimum salary.
db.students.aggregate([{$group:{_id:"$CITY",MAXFees:{$max:"$FEES"},minfees:{$min:"$FEES"}}}])

9. Display count of persons lives in Sydney city in your collection.
db.students.aggregate([{$match:{CITY:"Rajkot"}},{$count:"studentinrajkot"}])

10. Display average fees of Rajkot city.
db.students.aggregate([{$match:{CITY:"Rajkot"}},{$group:{_id:null,avgfees:{$avg:"$FEES"}}}])

11. Count the number of male and female students in each Department
db.students.aggregate([{$group:{_id:{Department:"$DEPARTMENT",gender:"$GENDER"},Count:{$sum:1}}}])

12. Find the total Fees collected from each Department.
db.students.aggregate([{$group:{_id:"$DEPARTMENT",Count:{$sum:"$FEES"}}}])

13. Find the minimum Fees paid by male and female students in each City.
db.students.aggregate([{$group:{_id:{City:"$CITY",gender:"$GENDER"},Count:{$min:"FEES"}}}])

14. Sort students by Fees in descending order and return the top 5
db.students.aggregate([{$sort:{FEES:-1}},{$limit:5}])

15. Group students by City and calculate the average Fees for each city, only including cities with more than 1
student.
db.students.aggregate([{$group:{_id:"$CITY",Count:{$avg:"$FEES"},sum:{$sum:1}}},{$match:{sum:{$gt:1}}}])
