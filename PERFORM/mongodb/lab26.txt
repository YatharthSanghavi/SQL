1. Count unique customers as per category.
db.Shopping_Order.aggregate([{$group:{_id:{CATEGORY:"$CATEGORY",CUSTOMER:"$CUSTOMER"}}},{$group:{_id:"$_id.CATEGORY",unique:{$sum:1}}}])

2. Find the highest order value of each Customer.
db.Shopping_Order.aggregate([{$group:{_id:"$CUSTOMER",TOTAL:{$max:"$Total"}}}])

3. Determine average order per product category.
db.Shopping_Order.aggregate([{$group:{_id:"$CATEGORY",AVGERAGE:{$avg:"$Total"}}}])

4. Group by customer and product,count total units ordered.
db.Shopping_Order.aggregate([{$group:{_id:{CUSTOMER:"$CUSTOMER",Product:"$PRODUCT"},total:{$sum:"$QTY"}}}])

5. Group by product, get average quantity and total revenue.
db.Shopping_Order.aggregate([{$group:{_id:"$PRODUCT",AVGQTY:{$avg:"$QTY"},Revenue:{$sum:"$Total"}}}])

6. As per day get total products sold.
db.Shopping_Order.aggregate([{$group:{_id:"$Date",totalsum:{$sum:"$Total"}}}])

7. Filter only those Customers with total spend > 5000.
db.Shopping_Order.aggregate([{$group:{_id:"$CUSTOMER"},{$match:{Total:{$gt:5000}}}}])

8. Average order value per status.
db.Shopping_Order.aggregate([{$group:{_id:"$Status",Avgvalue:{$avg:"$Total"}}}])

9. Find Top 3 Highest sold Product(revenue).
db.Shopping_Order.aggregate([{$group:{_id:"$PRODUCT",total:{$sum:"$Total"}}},{$sort:{total:-1}},{$limit:3}])

10. How many quantity of Product’s order is cancelled.
db.Shopping_Order.aggregate([{$match:{Status:"Cancelled"}},{$group:{_id:null,total:{$sum:"$QTY"}}}])

11. Count how many products each customer has bought across all categories
db.Shopping_Order.aggregate([{$group:{_id:{customer:"$CUSTOMER",category:"$CATEGORY"},qty:{$sum:"$QTY"}}}])

12. Get the Number of orders placed per Customer per status (Delivered,Shipped,etc..)
db.Shopping_Order.aggregate([{$group:{_id:{customer:"$CUSTOMER",status:"$Status"},qty:{$sum:1}}}])
