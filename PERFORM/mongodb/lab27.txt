2. Show project name and manager name. 
->
db.projects.aggregate([{
  $lookup:
  {
    from:"managers",
  	localField:"MANAGER_ID",
    foreignField:"MANAGER_ID",
    as:"ManagerTable"
  }
},{
  $project:{
    _id:0,
    PROJECT_NAME: 1,
  	"ManagerTable.NAME": 1
  }
}])
