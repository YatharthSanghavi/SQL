1. Display all distinct departments.
db.faculty.aggregate([{$group:{_id:"$DEPARTMENT"}}])

2. Count of faculty in each department.
db.faculty.aggregate([{$group:{_id:"$DEPARTMENT",COUNT:{$sum:1}}}])

3. Average salary of each department (without department name)
db.faculty.aggregate([{$group:{_id:"$DEPARTMENT",avg:{$avg:"$SALARY"}}},{$project:{_id:0,avg:1}}])

4.Get faculty with salary between 45000 and 50000
db.faculty.aggregate([{$match:{SALARY:{$gte:45000,$lte:50000}}}])

5. Departments having more than 2 faculty members
db.faculty.aggregate([{$group:{_id:"$DEPARTMENT",count:{$sum:1}}},{$match:{count:{$gt:2}}}])

6 Total salary per department where city is 'Rajkot' and salary > 45000
db.faculty.aggregate([{$match:{SALARY:{$gt:45000},CITY:"Rajkot"}},{$group:{_id:"$DEPARTMENT",TOTAL:{$sum:"$SALARY"}}}])

7. List faculties from departments CE or EC having more than 5 years of experience.
db.faculty.aggregate([{$match:{EXPERIENCE:{$gt:5},DEPARTMENT:{$in:["CE","EC"]}}}])

8. Display department and faculty name with highest experience
db.faculty.aggregate([{$sort:{EXPERIENCE:-1}},{$limit:1},{ $project: { _id: 0, FNAME: 1, DEPARTMENT: 1, EXPERIENCE: 1 }}])

9. Find second lowest salary
db.faculty.aggregate([{$sort:{SALARY:1}},{$limit:2},{$skip:1},{ $project: { _id: 0, FNAME: 1, DEPARTMENT: 1, SALARY: 1 }}])

10. City-wise average experience of male faculty
db.faculty.aggregate([{$match:{GENDER:"Male"}},{$group:{_id:"$CITY",AVG:{$avg:"$EXPERIENCE"}}}])

11. Sort by experience descending
db.faculty.aggregate([{$sort:{EXPERIENCE:-1}}])
